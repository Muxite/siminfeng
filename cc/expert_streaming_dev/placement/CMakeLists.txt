cmake_minimum_required(VERSION 3.18)
project(placement_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../llama-place" CACHE PATH "Pinned llama.cpp source")
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}/ggml" ggml)
add_executable(expert-crossover crossover.cpp)
target_link_libraries(expert-crossover PRIVATE ggml ggml-cpu)
target_compile_options(expert-crossover PRIVATE -Wall -Wextra -Wpedantic)
